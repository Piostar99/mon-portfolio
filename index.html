class RBACEngine:
    def check_permission(self, user, resource, action):
        roles = user.get_active_roles()
        for role in roles:
            if role.has_privilege(resource, action):
                return True
        raise PermissionDenied("Access Revoked")

from langchain_openai import ChatOpenAI
from langchain.vectorstores import PGVector

def generate_interview_question(cv_embeddings, job_description):
    vector_db = PGVector.from_existing_index(cv_embeddings)
    relevant_context = vector_db.similarity_search(job_description)
    llm = ChatOpenAI(model="gpt-4o")
    return llm.predict(relevant_context, job_description)
